The service file content:

  # Deployed to /etc/systemd/system/splitgate-admin.service by deploy.sh Stage 29.
  # Flask serves React SPA from /etc/splitgate/admin/ and REST API on ADMIN_PORT (default 8080).

  StandardOutput=null
  StandardError=append:/etc/splitgate/logs/admin-error.log
  Restart=on-failure
  RestartSec=5
  User=root
  
  [Install]
  WantedBy=multi-user.target
