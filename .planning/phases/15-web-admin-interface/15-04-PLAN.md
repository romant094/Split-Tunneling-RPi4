Insert the `admin)` case before the `*)` default case:

```bash
  admin)
    if $has_help; then
      echo "splitgate admin — manage the web admin interface service"
      echo ""
      echo "Usage: splitgate admin [status|start|stop]"
      echo ""
      echo "  status   Show whether splitgate-admin.service is running and the access URL"
      echo "  start    Start the admin service (sudo systemctl start splitgate-admin.service)"
      echo "  stop     Stop the admin service (sudo systemctl stop splitgate-admin.service)"
      echo ""
      echo "  Web UI accessible at http://192.168.1.254:\${ADMIN_PORT:-8080}"
      echo "  Password must be set in /etc/splitgate/admin.secret (chmod 600)"
      echo ""
      echo "Examples:"
      echo "  splitgate admin status"
      echo "  splitgate admin start"
      exit 0
    fi
    SUBCMD="${1:-status}"
    shift || true
    case "$SUBCMD" in
      status)
        STATE=$(systemctl is-active splitgate-admin.service 2>/dev/null || echo "unknown")
        PORT=$(grep -E '^ADMIN_PORT=' /etc/splitgate/vpn-gateway.env 2>/dev/null | cut -d= -f2 || echo "8080")
        echo "splitgate-admin.service: ${STATE}"
        echo "Access: http://192.168.1.254:${PORT}"
        ;;
      start)
        exec sudo systemctl start splitgate-admin.service
        ;;
      stop)
        exec sudo systemctl stop splitgate-admin.service
        ;;
      *)
        echo "Unknown admin subcommand: $SUBCMD" >&2
        echo "Usage: splitgate admin [status|start|stop]" >&2
        exit 1
        ;;
    esac
    ;;
```

Also update the default `*)` case Usage line to include admin:
```
echo "Usage: splitgate {status|watch|rollback|routing|update|admin} [args...]"
```

Note: no `exec` needed for the status sub-case (it just echoes). The start/stop sub-cases use exec for consistency with the exec-based pattern in the rest of the dispatcher. The `admin)` case uses `shift || true` to consume the sub-command arg before the sub-case switch.
