FLASK APP INIT:
  app = Flask(__name__)

AUTH DECORATOR (require_auth):
  @functools.wraps(f) wrapper that:
  1. Check cookie first: cookie_token = request.cookies.get('sg_session'); if cookie_token and cookie_token in _sessions: call f(*args, **kwargs) and return result directly
  2. Reads ADMIN_SECRET_PATH, strips whitespace -> stored_pw
  3. If no Basic Auth header present: return Response('Unauthorized', 401, {'WWW-Authenticate': 'Basic realm="splitgate"'})
  4. if not secrets.compare_digest(request.authorization.password.encode(), stored_pw.encode()): same 401 response
  5. else (Basic Auth succeeded): generate session_token = secrets.token_hex(16); _sessions.add(session_token); call f(*args, **kwargs) -> inner_result; resp = make_response(inner_result); resp.set_cookie('sg_session', session_token, httponly=True, samesite='Strict', path='/'); return resp

HELPER FUNCTIONS:
  read_routes_file(path) -> list[str]:
    If file not found return []. Open, read lines, skip lines starting with '#' or empty after strip, return stripped entries.
  write_routes_file(path, entries):
    Open path 'w', write each entry + '\n'. If entries is empty, write empty file.
  parse_env_file(path) -> dict[str, str]:
    If file not found return {}. Read lines, skip '#' comment lines and empty lines, split on first '=', return dict.
  write_env_file(path, data):
    Write 'KEY=VALUE\n' for each item.
  mask_value(key, value) -> str:
    If SECRET_KEY_RE.search(key): return '***' else return value.
  tail_file(path, n=200) -> list[str]:
    Open path, use deque(fh, maxlen=n), return list of stripped lines. Return [] on FileNotFoundError.

GET /api/routes/vpn and GET /api/routes/isp @require_auth:
  Return jsonify({'routes': read_routes_file(path)})

POST /api/routes/vpn and POST /api/routes/isp @require_auth methods=['POST']:
  body = request.get_json(silent=True) or {}
  cidr = body.get('cidr', '').strip()
  if not CIDR_RE.match(cidr): return jsonify({'error': 'Invalid CIDR'}), 400
  routes = read_routes_file(path)
  if cidr in routes: return jsonify({'error': 'Already exists'}), 409
  routes.append(cidr)
  write_routes_file(path, routes)
  Return 201 {'ok': True}

DELETE /api/routes/vpn and DELETE /api/routes/isp @require_auth methods=['DELETE']:
  body = request.get_json(silent=True) or {}
  cidr = body.get('cidr', '').strip()
  if not CIDR_RE.match(cidr): return 400
  routes = read_routes_file(path)
  write_routes_file(path, [r for r in routes if r != cidr])
  Return 200 {'ok': True}

GET /api/logs/watch @require_auth:
  SSE stream with stream_with_context. Generator:
    current_date = date.today()
    while True:
      log_path = f"{LOG_DIR}/watch-{current_date.strftime('%Y-%m-%d')}.log"
      try:
        with open(log_path, 'r') as fh:
          fh.seek(0, 2)  # seek to EOF - stream new lines only
          while True:
            today = date.today()
            if today != current_date:
              current_date = today
              break  # reopen new day file
            line = fh.readline()
            if line:
              yield f"data: {line.rstrip()}\n\n"
            else:
              time.sleep(0.1)
      except FileNotFoundError:
        yield f"data: [waiting for {log_path}]\n\n"
        time.sleep(2)
  Return Response(stream_with_context(generate()), content_type='text/event-stream',
                  headers={'Cache-Control': 'no-cache', 'X-Accel-Buffering': 'no'})

PUT /api/settings/env @require_auth methods=['PUT']:
  body = request.get_json(silent=True) or {}
  new_vars = body.get('vars', {})
  KEY_RE = re.compile(r'^[A-Z][A-Z0-9_]*$')
  if any(not KEY_RE.match(k) for k in new_vars): return jsonify({'error': 'Invalid key format'}), 400
  current = parse_env_file(ENV_PATH)
  current.update(new_vars)
  write_env_file(ENV_PATH, current)
  Return 200 {'ok': True}

GET /api/settings/secrets @require_auth:
  Parse AWG_CONF_PATH: read file, find all 'Key = Value' pairs using re.findall(r'^\s*(\w+)\s*=\s*(.+)$', content, re.MULTILINE)
  Return jsonify({'vars': {k: mask_value(k, v.strip()) for k, v in pairs}})

PUT /api/settings/secrets @require_auth methods=['PUT']:
  body = request.get_json(silent=True) or {}
  new_vars = body.get('vars', {})
  content = open(AWG_CONF_PATH).read()
  for key, val in new_vars.items():
    content = re.sub(rf'^(\s*{re.escape(key)}\s*=\s*)(.+)$', rf'\g<1>{val}', content, flags=re.MULTILINE)
  with open(AWG_CONF_PATH, 'w') as fh: fh.write(content)
  os.chmod(AWG_CONF_PATH, 0o600)
  Return 200 {'ok': True}

POST /api/settings/password @require_auth methods=['POST']:
  body = request.get_json(silent=True) or {}
  new_pw = body.get('password', '').strip()
  if not new_pw: return jsonify({'error': 'Password required'}), 400
  with open(ADMIN_SECRET_PATH, 'w') as fh: fh.write(new_pw + '\n')
  os.chmod(ADMIN_SECRET_PATH, 0o600)
  Return 200 {'ok': True}

POST /api/auth/logout methods=['POST']:
  token = request.cookies.get('sg_session')
  if token:
    _sessions.discard(token)
  resp = make_response(jsonify({'ok': True}))
  resp.set_cookie('sg_session', '', expires=0, httponly=True, samesite='Strict', path='/')
  return resp

POST /api/settings/rollback @require_auth methods=['POST']:
  body = request.get_json(silent=True) or {}
  if body.get('confirmation') != 'ROLLBACK': return jsonify({'error': 'Send body: {"confirmation": "ROLLBACK"}'}), 400
  # Run in background thread - rollback removes admin itself
  def do_rollback():
    subprocess.run([ROLLBACK_SH], timeout=60)
  threading.Thread(target=do_rollback, daemon=True).start()
  Return jsonify({'message': 'Rollback initiated — service will stop'}), 200

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Browser (LAN) → Flask :8080 | HTTP Basic Auth gate; unauthenticated requests must be rejected |
| Flask process (root) → filesystem | Writes to /etc/splitgate/ files and /etc/amnezia/ — root-only paths |
| Flask → subprocess | systemctl, journalctl, routing.sh, vpn-rollback.sh — all as root |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-15-01 | Spoofing | HTTP Basic Auth | mitigate | secrets.compare_digest prevents timing attacks; 401 on missing or wrong creds |
| T-15-02 | Tampering | CIDR route file writes | mitigate | CIDR_RE validates input before any file open('w'); 400 on mismatch |
| T-15-03 | Tampering | env file key injection | mitigate | KEY_RE = re.compile(r'^[A-Z][A-Z0-9_]*$') validates all keys before write_env_file |
| T-15-04 | Tampering | subprocess command injection | mitigate | All subprocess.run calls use list argv; shell=True forbidden; service names validated against MANAGED_SERVICES allowlist |
| T-15-05 | Tampering | AWG conf write | mitigate | re.sub replaces only existing keys; os.chmod(0o600) after write |
| T-15-06 | Information Disclosure | VPN keys in browser | accept | User accepted risk per D-14; LAN-only + Basic Auth gate is sufficient |
| T-15-07 | Information Disclosure | admin.secret exposure | mitigate | File mode 600 root:root; never echoed in response or logs |
| T-15-08 | Elevation of Privilege | Flask runs as root | accept | Required per D-11 (systemctl + /etc/splitgate/ writes); no privilege drop possible |
| T-15-09 | Denial of Service | SSE connection held open | mitigate | time.sleep(0.1) prevents busy-wait; client disconnect raises GeneratorExit cleanly |
| T-15-SC | Tampering | pip/npm installs in this plan | mitigate | No installs in Plan 01; Flask assumed pre-installed via python3-flask apt package |
</threat_model>
